Практическое задание 2. Вычисления через TCP (Вариант 3)

Задача: Реализовать поиск площади трапеции. Параметры вводятся на клиенте, сервер проводит вычисления и возвращает результат.

Сервер (server.py)

import socket

server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server_socket.bind(('127.0.0.1', 9090))
server_socket.listen(1)

conn, addr = server_socket.accept()

data = conn.recv(1024).decode('utf-8')
a, b, h = map(float, data.split(','))

area = ((a + b) / 2) * h

conn.send(str(area).encode('utf-8'))

conn.close()
server_socket.close()

Клиент (client.py)

import socket

client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
client_socket.connect(('127.0.0.1', 9090))

a = input("a: ")
b = input("b: ")
h = input("h: ")

client_socket.send(f"{a},{b},{h}".encode('utf-8'))

result = client_socket.recv(1024).decode('utf-8')
print(result)

client_socket.close()
